cmake_minimum_required(VERSION 3.23)
project(wemeet_desktop_capture LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(PkgConfig REQUIRED)
pkg_check_modules(CAPTURE REQUIRED IMPORTED_TARGET libpipewire-0.3 gio-unix-2.0 egl glesv2 libdrm x11 xext)
find_package(Threads REQUIRED)
find_package(Qt5Core REQUIRED)
find_package(Qt5Widgets REQUIRED)
add_library(wemeet-capture SHARED capture-hook.cpp capture-overlay.cpp)
target_link_libraries(wemeet-capture PRIVATE PkgConfig::CAPTURE Threads::Threads ${CMAKE_DL_LIBS})
target_include_directories(wemeet-capture PRIVATE ${Qt5Widgets_INCLUDE_DIRS})
target_compile_definitions(wemeet-capture PRIVATE ${Qt5Widgets_DEFINITIONS} QT_NO_VERSION_TAGGING)
target_compile_options(wemeet-capture PRIVATE -Wall -Wextra)
enable_testing()
add_executable(capture-tests capture-tests.cpp)
target_link_libraries(capture-tests PRIVATE PkgConfig::CAPTURE)
add_test(NAME capture-frame-boundaries COMMAND capture-tests)
add_test(NAME capture-preload-child COMMAND ${CMAKE_COMMAND} -E env
  LD_PRELOAD=$<TARGET_FILE:wemeet-capture> LD_BIND_NOW=1 ${CMAKE_COMMAND} -E echo "preloaded child started")
install(TARGETS wemeet-capture LIBRARY DESTINATION lib)
